### **БЛОК E: Семейный тариф — механизм подключения профилей**

**Задача:** глава семьи может пригласить до 4 членов через ссылку. У каждого свой профиль (тренировки, замеры, меню). Оплата — общая.

**Миграция 0029:**

```sql
CREATE TABLE IF NOT EXISTS family_members (
  id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  owner_id UUID NOT NULL REFERENCES users(id) ON DELETE CASCADE,
  member_id UUID NOT NULL REFERENCES users(id) ON DELETE CASCADE,
  invited_at TIMESTAMPTZ DEFAULT now(),
  accepted_at TIMESTAMPTZ,
  UNIQUE(owner_id, member_id)
);

CREATE INDEX IF NOT EXISTS idx_family_owner ON family_members(owner_id);
CREATE INDEX IF NOT EXISTS idx_family_member ON family_members(member_id);

-- Ограничение: не более 3 членов (4 профиля = 1 глава + 3 приглашённых)
-- Проверяем в коде, не констрейнтом (иначе миграция может упасть при seed)

CREATE TABLE IF NOT EXISTS family_invites (
  id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  owner_id UUID NOT NULL REFERENCES users(id) ON DELETE CASCADE,
  invite_code TEXT NOT NULL UNIQUE,
  created_at TIMESTAMPTZ DEFAULT now(),
  expires_at TIMESTAMPTZ NOT NULL,
  used_by UUID REFERENCES users(id),
  used_at TIMESTAMPTZ
);

CREATE INDEX IF NOT EXISTS idx_invites_code ON family_invites(invite_code);

ALTER TABLE family_members ENABLE ROW LEVEL SECURITY;
ALTER TABLE family_invites ENABLE ROW LEVEL SECURITY;

CREATE POLICY "family_members_read_own" ON family_members
  FOR SELECT USING (
    auth.uid()::text = owner_id::text OR 
    auth.uid()::text = member_id::text
  );

CREATE POLICY "family_invites_own" ON family_invites
  FOR ALL USING (auth.uid()::text = owner_id::text);

NOTIFY pgrst, 'reload schema';
```
